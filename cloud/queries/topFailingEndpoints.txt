/*
Shows the HTTP endpoints with the highest error counts, sorted by frequency.
*/

ContainerAppConsoleLogs_CL
| where ContainerAppName_s == "backend"
| where TimeGenerated > ago(24h)
| where Log_s matches regex @'"\s(4\d{2}|5\d{2})\s'
| extend Path = extract(@'"(GET|POST|PUT|DELETE|PATCH)\s(\S+)', 2, Log_s)
| extend Status = extract(@'HTTP/1\.\d"\s(\d{3})', 1, Log_s)
| summarize FailCount = count() by Path, Status
| order by FailCount desc
